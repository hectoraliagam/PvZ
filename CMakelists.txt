cmake_minimum_required(VERSION 3.16)

project(pvz 
	VERSION 0.1 
	LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ===== Build type =====
option(PRODUCTION_BUILD "Build for production" OFF)

if(PRODUCTION_BUILD)
	add_compile_definitions(PRODUCTION_BUILD=1 DEVELOPMENT_BUILD=0)
else()
	add_compile_definitions(PRODUCTION_BUILD=0 DEVELOPMENT_BUILD=1)
endif()

# ===== SFML =====
find_package(SFML CONFIG REQUIRED COMPONENTS
	Graphics
	Window
	System
	Audio
)

# ===== Sources =====
file(GLOB_RECURSE PVZ_SOURCES CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"	
)

add_executable(pvz_game ${PVZ_SOURCES})

# ===== Include dirs =====
target_include_directories(pvz_game PRIVATE
	"${CMAKE_CURRENT_SOURCE_DIR}/include"
)

# ===== Compile options =====
if(MSVC)
	target_compile_options(pvz_game PRIVATE /W4 /permissive-)
else()
	target_compile_options(pvz_game PRIVATE -Wall -Wextra --Wpedantic)
endif()

# ===== Resources path =====
if(PRODUCTION_BUILD)
	target_compile_definitions(pvz_game PRIVATE 
		RESOURCES_PATH="./resources/"
	)
else()
	target_compile_definitions(pvz_game PRIVATE
		RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/"
	)
endif()

# ===== Link =====
target_link_libraries(pvz_game PRIVATE
	SFML::Graphics
	SFML::Window
	SFML::System
	SFML::Audio
)
